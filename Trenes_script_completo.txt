-- [Tablas Auxiliares]
CREATE TABLE  Tipos_Vagones(
   Id_Tipo_Vagon INT IDENTITY(1,1) NOT NULL,
   Descripcion NVARCHAR(50),
   CONSTRAINT pk_Tipo_Vagon PRIMARY KEY (Id_Tipo_Vagon)
);
CREATE TABLE  Tipos_Mantenimientos(
	Id_Tipo_Mantenimiento INT IDENTITY(1,1) NOT NULL,
	Descripcion NVARCHAR(50),
	
	CONSTRAINT pk_Tipo_Mantenimientos PRIMARY KEY (Id_Tipo_Mantenimiento)
);
CREATE TABLE  Tipos_Documento(
   Id_Tipo_Doc INT IDENTITY(1,1) NOT NULL,
   Descripcion NVARCHAR(50),
  
   CONSTRAINT pk_Tipos_Documento PRIMARY KEY (Id_Tipo_Doc)
);
CREATE TABLE  Tipos_Contacto(
   Id_Tipo_Contacto INT IDENTITY(1,1) NOT NULL,
   Descripcion NVARCHAR(50),
   CONSTRAINT pk_Tipos_Contacto PRIMARY KEY (Id_Tipo_Contacto)
);
CREATE TABLE  Roles(
	Id_Rol INT IDENTITY(1,1) NOT NULL,
	Descripcion NVARCHAR(50),
	CONSTRAINT pk_Roles PRIMARY KEY (Id_Rol)
);
CREATE TABLE  Tipos_Repuestos(
	Id_Tipo_Repuesto INT IDENTITY(1,1) NOT NULL,
	Descripcion_Tipo NVARCHAR(50),
	CONSTRAINT pk_Tipos_Repuesto PRIMARY KEY (Id_Tipo_Repuesto)
);
CREATE TABLE  Paises(
	Id_Pais INT IDENTITY(1,1) NOT NULL,
	Descripcion NVARCHAR(50),
	CONSTRAINT pk_Paises PRIMARY KEY (Id_Pais)
);
CREATE TABLE  Provincias(
   Id_Provincia INT IDENTITY(1,1) NOT NULL,
   Descripcion NVARCHAR(50),
   Id_Pais INT NOT NULL,
   CONSTRAINT pk_Provincias PRIMARY KEY (id_provincia),
   CONSTRAINT fk_Barrios_Paises FOREIGN KEY (Id_Pais)
       REFERENCES Paises (Id_Pais)
);
CREATE TABLE  Localidades(
	Id_Localidad INT IDENTITY(1,1) NOT NULL,
	Descripcion NVARCHAR(50),
	Id_Provincia INT NOT NULL,
	CONSTRAINT pk_Localidades PRIMARY KEY (Id_Localidad),
	CONSTRAINT fk_Barrios_Provincias FOREIGN KEY (Id_Provincia)
		REFERENCES Provincias (Id_Provincia)
);
CREATE TABLE  Barrios(
   Id_Barrio INT IDENTITY(1,1) NOT NULL,
   Descripcion NVARCHAR(50),
   Id_Localidad INT NOT NULL,
   CONSTRAINT pk_Barrios PRIMARY KEY (Id_Barrio),
   CONSTRAINT fk_Barrios_Localidades FOREIGN KEY (Id_Localidad)
       REFERENCES Localidades (Id_Localidad)
);
CREATE TABLE  Estados(
	Id_Estado INT IDENTITY(1,1) NOT NULL,
	Descripcion NVARCHAR(50),
	CONSTRAINT pk_Estados PRIMARY KEY (Id_Estado)
);
CREATE TABLE  Formas_Pago(
	Id_Forma_Pago INT IDENTITY(1,1) NOT NULL,
	Descripcion NVARCHAR(50),
	CONSTRAINT pk_Formas_Pago PRIMARY KEY (Id_Forma_Pago)
);
CREATE TABLE  Tipos_Movimientos(
	Id_Tipo_Movimiento INT IDENTITY(1,1) NOT NULL,
	Descripcion NVARCHAR(50),
	CONSTRAINT pk_Tipos_Movimiento PRIMARY KEY (Id_Tipo_Movimiento)
);
	
CREATE TABLE  Servicios(
	Id_Servicio INT IDENTITY(1,1) NOT NULL,
	Descripcion NVARCHAR(50),
	Valor_servicio DECIMAL(18,2),
	CONSTRAINT pk_Servicios PRIMARY KEY (Id_Servicio)
);
CREATE TABLE  Choferes(
   Id_Chofer INT IDENTITY(1,1) NOT NULL,
   Id_Empleado INT NOT NULL,
   Activo BIT,
   CONSTRAINT pk_Choferes PRIMARY KEY (Id_Chofer)
);
-- [Tablas Principales]
CREATE TABLE  Trenes(
	Id_Tren INT IDENTITY(1,1) NOT NULL,
	Nro_unidad NVARCHAR(50),
	Modelo NVARCHAR(50),
	Capacidad INT,
	Id_Estado INT NOT NULL,
	CONSTRAINT pk_Trenes PRIMARY KEY (Id_Tren),
	CONSTRAINT fk_Estados_Trenes FOREIGN KEY (Id_Estado)
		REFERENCES Estados (Id_Estado)
);
	
CREATE TABLE  Vagones (
   Id_Vagon INT IDENTITY(1, 1) NOT NULL,   
   Id_Tipo_Vagon INT NOT NULL,
   Id_Servicio INT NOT NULL,
   Capacidad INT,
   Cantidad_asientos INT,
  
   CONSTRAINT pk_Vagones PRIMARY KEY (Id_Vagon),
   CONSTRAINT fk_Tipos_Vagones_Vagones FOREIGN KEY (Id_Tipo_Vagon)
       REFERENCES Tipos_Vagones (Id_Tipo_Vagon),
   CONSTRAINT fk_Servicios_Vagones FOREIGN KEY (Id_Servicio)
       REFERENCES Servicios (Id_Servicio)
);
CREATE TABLE  TrenesxVagones(
   Id_Trenes_Vagones  INT IDENTITY(1,1) NOT NULL,
   Id_Tren INT NOT NULL,
   Id_Vagon INT NOT NULL,
   CONSTRAINT pk_TrenesxVagones PRIMARY KEY (Id_Trenes_Vagones),
   CONSTRAINT fk_Vagones_TrenesxVagones FOREIGN KEY (Id_Vagon)
      REFERENCES Vagones (Id_Vagon),
  CONSTRAINT fk_Trenes_TrenesxVagones FOREIGN KEY(Id_Tren)
      REFERENCES Trenes (Id_Tren)
);
CREATE TABLE  Estaciones(
   Id_Estacion  INT IDENTITY(1,1) NOT NULL,
   Ubicacion NVARCHAR(50),
   Nombre NVARCHAR(50),
   Plataforma INT,
   Id_Barrio INT NOT NULL,
   CONSTRAINT pk_Estaciones PRIMARY KEY (Id_Estacion),
   CONSTRAINT fk_Barrios_Estaciones FOREIGN KEY (Id_Barrio)
       REFERENCES Barrios (Id_Barrio)
);
CREATE TABLE  Recorridos(
   Nro_Linea  INT IDENTITY(1,1) NOT NULL,
   Descripcion NVARCHAR(200),
   CONSTRAINT pk_Recorridos PRIMARY KEY (Nro_Linea)
);
CREATE TABLE  Estaciones_Recorridos(
	Id_Estacion_Recorrido  INT IDENTITY(1,1) NOT NULL,
	Nro_Linea INT NOT NULL,
	Id_Estacion INT NOT NULL,
	
	CONSTRAINT pk_Estaciones_Recorridos PRIMARY KEY (Id_Estacion_Recorrido),
	CONSTRAINT fk_Recorridos_Estaciones_Recorridos FOREIGN KEY (Nro_Linea)
		REFERENCES Recorridos (Nro_Linea),
	CONSTRAINT fk_Estaciones_Estaciones_Recorridos FOREIGN KEY (Id_Estacion)
		REFERENCES Estaciones (Id_Estacion)
);
	
CREATE TABLE   Empleados(
	Id_Empleado INT IDENTITY(1,1) NOT NULL,
	Id_Rol INT NOT NULL,
	Nombre NVARCHAR(50),
	Apellido NVARCHAR(50),
	NroDoc INT,
	Id_Tipo_Doc INT NOT NULL,
	Id_Barrio INT NOT NULL,
	Calle NVARCHAR(50),
	Numero INT,
	
	CONSTRAINT pk_Empleados PRIMARY KEY (Id_Empleado),
	CONSTRAINT fk_Rol FOREIGN KEY (Id_Rol)
		REFERENCES roles (Id_Rol),
	CONSTRAINT fk_Tipos_Documentos FOREIGN KEY (Id_Tipo_Doc)
		REFERENCES Tipos_Documento (Id_Tipo_Doc),
	CONSTRAINT fk_barrios FOREIGN KEY (Id_Barrio)
		REFERENCES Barrios (Id_Barrio)
);
CREATE TABLE  Clientes(
   Id_Cliente INT IDENTITY(1,1) NOT NULL,
   NroDoc INT,
   Id_Tipo_Doc INT,
   Id_Barrio INT,
   Calle NVARCHAR(50),
   Numero INT,
   CONSTRAINT pk_Clientes PRIMARY KEY (Id_Cliente),
   CONSTRAINT fk_barrios_clientes FOREIGN KEY (Id_Barrio)
       REFERENCES Barrios (Id_Barrio),
   CONSTRAINT fk_Tipos_Documento_clientes FOREIGN KEY (Id_Tipo_Doc)
       REFERENCES Tipos_Documento (Id_Tipo_Doc)
);
CREATE TABLE  Tickets(
	Id_Ticket INT IDENTITY(1,1) NOT NULL,
	Fecha DATETIME2,
	Id_Empleado INT,
	Id_Cliente INT,
	CONSTRAINT pk_Tickets PRIMARY KEY (id_ticket),
	CONSTRAINT fk_Clientes_Tickets FOREIGN KEY (Id_Cliente)
		REFERENCES Clientes (Id_Cliente),
	CONSTRAINT fk_Empleado_Tickets FOREIGN KEY (Id_Empleado)
		REFERENCES Empleados (Id_Empleado)
);
CREATE TABLE  Repuestos(
   Id_Repuesto INT IDENTITY(1,1) NOT NULL,
   Id_Tipo_Repuesto INT,
   Nombre NVARCHAR(50),
   Material NVARCHAR(50),
   Precio_Unit INT,
   CONSTRAINT pk_Repuestos PRIMARY KEY (Id_Repuesto),
   CONSTRAINT fk_tipo_repuesto_Repuestos FOREIGN KEY (Id_Tipo_Repuesto)
       REFERENCES Tipos_Repuestos(Id_Tipo_Repuesto)
);
CREATE TABLE  Stocks(
	Id_Stock INT IDENTITY(1,1) NOT NULL,
	Id_Repuesto INT,
	Cantidad INT,
	CONSTRAINT pk_Stock PRIMARY KEY (Id_Stock),
	CONSTRAINT fk_Repuestos_Stock FOREIGN KEY (Id_Repuesto)
		REFERENCES Repuestos (Id_Repuesto)
);
CREATE TABLE  Facturas (
   Nro_Factura INT IDENTITY(1,1) NOT NULL,
   Id_Proveedor INT,
   Id_Empleado INT,
   Fecha DATETIME2,
   Monto INT,
   CONSTRAINT pk_Facturas PRIMARY KEY  (Nro_Factura),
   CONSTRAINT fk_Empleado_Facturas FOREIGN KEY (Id_Empleado)
       REFERENCES empleados (Id_Empleado)
);
CREATE TABLE  Proveedores(
   Id_Proveedor INT IDENTITY(1,1) NOT NULL,
   Nombre NVARCHAR(50),
   Cuil INT,
   Id_Barrio INT,
   Calle NVARCHAR(50),
   Numero INT,
   CONSTRAINT pk_proveedores PRIMARY KEY (Id_Proveedor),
   CONSTRAINT fk_barrio_proveedores FOREIGN KEY (Id_Barrio)
       REFERENCES barrios (Id_Barrio)
);
CREATE TABLE  Contactos_Proveedor(
   Id_Contacto_Prov INT IDENTITY(1,1) NOT NULL,
   Id_Proveedor INT NOT NULL,
   Id_Tipo_Contacto INT NOT NULL,
   Contactos NVARCHAR(50),
   CONSTRAINT pk_Contactos_Proveedor PRIMARY KEY (Id_Contacto_Prov),
   CONSTRAINT fk_Proveedor_Contactos_Proveedor FOREIGN KEY (Id_Proveedor) REFERENCES Proveedores (id_proveedor),
   CONSTRAINT fk_Tipos_Contactos_Contactos_Proveedor FOREIGN KEY (Id_Tipo_Contacto)
       REFERENCES Tipos_Contacto (Id_Tipo_Contacto)
);
CREATE TABLE  Trenes_Recorridos_Estaciones(
   Id_Tren_Recorrido_Est INT IDENTITY(1,1) NOT NULL,
   Id_Trenes_Vagones INT NOT NULL,
   Nro_Linea INT NOT NULL,
   Hora_Partida TIME,
   Hora_Llegada TIME,
   Id_Estacion INT NOT NULL,
   Hora_Estacion TIME,
   Id_Chofer INT NOT NULL,
   Id_Empleado INT NOT NULL,
   CONSTRAINT pk_Trenes_Recorridos_Estaciones PRIMARY KEY (Id_Tren_Recorrido_Est),
   CONSTRAINT fk_TrenesxVagones_Trenes_Recorridos_Estaciones FOREIGN KEY (Id_Trenes_Vagones)
       REFERENCES TrenesxVagones (Id_Trenes_Vagones),
   CONSTRAINT fk_Recorridos_Trenes_Recorridos_Estaciones FOREIGN KEY (Nro_Linea)
       REFERENCES Recorridos (Nro_Linea),
   CONSTRAINT fk_Estaciones_Trenes_Recorridos_Estaciones FOREIGN KEY (Id_Estacion)
       REFERENCES Estaciones (Id_Estacion),
   CONSTRAINT fk_Choferes_Trenes_Recorridos_Estaciones FOREIGN KEY (Id_Chofer)
       REFERENCES Choferes (Id_Chofer),
   CONSTRAINT fk_Empleados_Trenes_Recorridos_Estaciones FOREIGN KEY (Id_Empleado)
       REFERENCES Empleados (Id_Empleado)
); 
CREATE TABLE Detalle_Tickets(
   Id_Det_Ticket INT IDENTITY(1,1) NOT NULL,
   Id_Trenes_Vagones INT NOT NULL,
   Id_Tren_Recorrido_Est INT NOT NULL,
   Id_Ticket INT NOT NULL,
   Precio DECIMAL(18,2),
   CONSTRAINT pk_Detalle_Tickets PRIMARY KEY (Id_Det_Ticket ),
   CONSTRAINT fk_Detalle_Tickets_Tickets FOREIGN KEY (Id_Ticket)
   REFERENCES Tickets (Id_Ticket),
   CONSTRAINT fk_Detalle_Tickets_Trenes_Recorridos_Estaciones FOREIGN KEY (Id_Tren_Recorrido_Est)
   REFERENCES Trenes_Recorridos_Estaciones (Id_Tren_Recorrido_Est),
   CONSTRAINT fk_Detalle_Tickets_Trenes_Vagones FOREIGN KEY (Id_Trenes_Vagones)
   REFERENCES TrenesxVagones (Id_Trenes_Vagones)
);
CREATE TABLE Tickets_x_Formapago(
   Id_Tickets_x_Fp INT IDENTITY(1,1),
   Id_Forma_Pago INT NOT NULL,
   Id_Ticket INT,
   CONSTRAINT pk_TicketsxFormapago PRIMARY KEY (Id_Tickets_x_Fp),
   CONSTRAINT fk_TicketsxFormaPago_Forma_Pagos FOREIGN KEY (Id_Forma_Pago)
   REFERENCES Formas_Pago (Id_Forma_Pago),
   CONSTRAINT fk_TicketsxFormaPago_Tickets FOREIGN KEY (Id_Ticket)
   REFERENCES Tickets (Id_Ticket)
);
CREATE TABLE Facturas_x_Formapago(
   Id_Fac_x_Formp INT IDENTITY(1,1),
   Id_Forma_Pago INT NOT NULL,
   Nro_Factura INT,
   CONSTRAINT pk_FacturasxFormapago PRIMARY KEY (Id_Fac_x_Formp),
   CONSTRAINT fk_FacturasxFormaPago_Forma_Pagos FOREIGN KEY (Id_Forma_Pago)
   REFERENCES Formas_Pago (Id_Forma_Pago),
   CONSTRAINT fk_FacturasxFormaPago_Facturas FOREIGN KEY (Nro_Factura)
   REFERENCES Facturas (Nro_Factura)
);
CREATE TABLE  Mantenimientos(
   Id_Mantenimiento INT IDENTITY(1,1) NOT NULL,
   Id_Tipo_Mantenimiento INT NOT NULL,
   Fecha DATETIME2,
   Id_Empleado INT NOT NULL,
   Id_Tren INT NOT NULL,
   CONSTRAINT pk_Mantenimientos PRIMARY KEY (Id_Mantenimiento),
   CONSTRAINT fk_Tipos_Mantenimientos_Mantenimientos FOREIGN KEY (Id_Tipo_Mantenimiento)
       REFERENCES Tipos_Mantenimientos (Id_Tipo_Mantenimiento),
   CONSTRAINT fk_Empleados_Mantenimientos FOREIGN KEY (Id_Empleado)
       REFERENCES Empleados (Id_Empleado),
   CONSTRAINT fk_Trenes_Mantenimientos FOREIGN KEY (Id_Tren)
       REFERENCES Trenes (Id_Tren)
);
CREATE TABLE Stock_Movimientos(
   Id_Stock_Movimiento int identity (1,1) NOT NULL,
   Id_Tipo_Movimiento int not null,
   Id_Stock int not null,
   Fecha datetime,
   Cantidad int,
   CONSTRAINT pk_Stock_Movimientos primary key (Id_Stock_Movimiento),
   Constraint fk_Stock_Movimientos_Tipo_Movimiento FOREIGN KEY (Id_Tipo_movimiento)
   REFERENCES Tipos_Movimientos (Id_Tipo_Movimiento),
       Constraint fk_Stock_Stock_Movimientos FOREIGN KEY (Id_Stock)
   REFERENCES Stocks (Id_Stock)
);
CREATE TABLE Detalles_Facturas (
   Id_Detalle_Factura int identity (1,1) NOT NULL,
   Nro_Factura int not null,
   Id_Repuesto int not null,
   Cantidad int,
   Precio decimal(18,2),
   CONSTRAINT pk_Detalles_Facturas primary key (Id_Detalle_Factura),
   Constraint fk_Detalles_Facturas_Facturas FOREIGN KEY (Nro_Factura)
   REFERENCES Facturas (Nro_Factura),
   Constraint fk_Detalles_Facturas_Repuestos FOREIGN KEY (Id_Repuesto)
   REFERENCES Repuestos (Id_Repuesto)
);
CREATE TABLE Contactos_Clientes (
   Id_Contacto_Cliente int identity (1,1) NOT NULL,
   Id_Cliente int not null,
   Id_Tipo_Contacto int not null,
   Contacto varchar(150),
   CONSTRAINT pk_Contactos_Clientes primary key (Id_Contacto_Cliente),
   Constraint fk_Contactos_Clientes_Clientes FOREIGN KEY (Id_Cliente)
   REFERENCES Clientes (Id_Cliente),
   Constraint fk_Contactos_Clientes_Tipos_Contactos FOREIGN KEY (Id_Tipo_Contacto)
   REFERENCES Tipos_Contacto (Id_Tipo_Contacto)
);
CREATE TABLE Contactos_Empleados (
   Id_Contacto_Empleado int identity (1,1) NOT NULL,
   Id_Empleado int not null,
   Id_Tipo_Contacto int not null,
   Contacto varchar(150),
   CONSTRAINT pk_Contactos_Empleados primary key (Id_Contacto_Empleado),
   Constraint fk_Contactos_Empleados_Empleados FOREIGN KEY (Id_Empleado)
   REFERENCES Empleados (Id_Empleado),
   Constraint fk_Contactos_Empleados_Tipos_Contactos FOREIGN KEY (Id_Tipo_Contacto)
   REFERENCES Tipos_Contacto (Id_Tipo_Contacto)
);

CREATE PROCEDURE dbo.PA_ReporteAvanzadoLineaTren
    @Nro_Linea INT
AS
BEGIN
    SET NOCOUNT ON;

    IF NOT EXISTS (SELECT 1 FROM Recorridos WHERE Nro_Linea = @Nro_Linea)
    BEGIN
        RAISERROR(N'El n�mero de l�nea especificado no existe.', 16, 1);
        RETURN;
    END

    SELECT
        r.Nro_Linea   AS NumeroLinea,
        r.Descripcion AS DescripcionRecorrido,
        t.Nro_unidad  AS UnidadTren,
        t.Modelo      AS ModeloTren,
        (SELECT COUNT(*) FROM Mantenimientos m WHERE m.Id_Tren = t.Id_Tren)     AS TotalMantenimientosHistoricos,
        (SELECT MAX(m.Fecha) FROM Mantenimientos m WHERE m.Id_Tren = t.Id_Tren) AS UltimaFechaMantenimiento,
        COUNT(DISTINCT er.Id_Estacion) AS CantidadEstacionesEnRecorrido,
        COUNT(DISTINCT txv.Id_Vagon)   AS CantidadVagonesAsignados,
        SUM(ISNULL(v.Capacidad, 0))    AS CapacidadTotalVagones,
        CASE
            WHEN t.Id_Estado = 1 THEN 'Operativo / Activo'
            WHEN t.Id_Estado = 2 THEN 'Fuera de Servicio'
            WHEN t.Id_Estado = 3 THEN 'En Mantenimiento'
            ELSE 'Estado No Definido'
        END AS EstadoOperativoTren
    FROM Recorridos r
    INNER JOIN Estaciones_Recorridos er ON r.Nro_Linea = er.Nro_Linea
    INNER JOIN Trenes_Recorridos_Estaciones tre ON r.Nro_Linea = tre.Nro_Linea AND er.Id_Estacion = tre.Id_Estacion
    INNER JOIN TrenesxVagones txv ON tre.Id_Trenes_Vagones = txv.Id_Trenes_Vagones
    INNER JOIN Trenes t ON txv.Id_Tren = t.Id_Tren
    LEFT JOIN Vagones v ON txv.Id_Vagon = v.Id_Vagon
    WHERE r.Nro_Linea = @Nro_Linea
    GROUP BY r.Nro_Linea, r.Descripcion, t.Nro_unidad, t.Modelo, t.Id_Estado, t.Id_Tren;
END;

GO

CREATE PROCEDURE dbo.P_ObtenerInformacionTrenPorLinea

@Nro_Linea INT

AS
BEGIN
IF NOT EXISTS (SELECT 1 FROM Recorridos WHERE Nro_Linea = @Nro_Linea)
   BEGIN
       RAISERROR('El número de línea especificado no existe.', 16, 1);
RETURN;
END
SELECT
r.Nro_Linea AS [Linea],
r.Descripcion AS [Recorrido],
t.Nro_unidad AS [Tren],
t.Modelo AS [Modelo_Tren],
(SELECT COUNT(*) FROM Mantenimientos m WHERE m.Id_Tren = t.Id_Tren) AS [Total_Mantenimientos_Historicos],
(SELECT MAX(m.Fecha) FROM Mantenimientos m WHERE m.Id_Tren = t.Id_Tren) AS [Ultimo_Mantenimiento],
COUNT(DISTINCT er.Id_Estacion) AS [Cantidad_Estaciones],
COUNT(DISTINCT txv.Id_Vagon) AS [Vagones_Asignados],
SUM(ISNULL(v.Capacidad, 0)) AS [Capacidad_Vagones],
CASE
       WHEN t.Id_Estado = 1 THEN 'Operativo / Activo'
       WHEN t.Id_Estado = 2 THEN 'Fuera de Servicio'
       WHEN t.Id_Estado = 3 THEN 'En Mantenimiento'
       ELSE 'Estado No Definido'
END AS [Estado_Operativo]
FROM Recorridos r
INNER JOIN Estaciones_Recorridos er ON r.Nro_Linea = er.Nro_Linea
INNER JOIN Trenes_Recorridos_Estaciones tre ON r.Nro_Linea = tre.Nro_Linea AND er.Id_Estacion = tre.Id_Estacion
INNER JOIN TrenesxVagones txv ON tre.Id_Trenes_Vagones = txv.Id_Trenes_Vagones
INNER JOIN Trenes t ON txv.Id_Tren = t.Id_Tren
LEFT JOIN Vagones v ON txv.Id_Vagon = v.Id_Vagon
WHERE r.Nro_Linea = @Nro_Linea
GROUP BY r.Nro_Linea,r.Descripcion,t.Nro_unidad,t.Modelo,t.Id_Estado,T.Id_Tren
END;

CREATE  FUNCTION dbo.fn_Historial_Compras_Cliente_DNI
(
    @NroDoc int
)
RETURNS TABLE
AS
RETURN
(
    SELECT
        c.Id_Cliente,
        c.NroDoc,
        td.Descripcion 'Tipo_Documento',
        t.Id_Ticket,
        t.Fecha 'Fecha_Compra',
        dt.Id_Det_Ticket,
        r.Nro_Linea,
        r.Descripcion 'Recorrido',
        es.Nombre 'Estacion',
        tre.Nro_unidad 'Tren',
        tr.Hora_Partida,
        tr.Hora_Llegada,
        dt.Precio
    FROM Clientes c
    left JOIN Tipos_Documento td ON td.Id_Tipo_Doc = c.Id_Tipo_Doc
    INNER JOIN Tickets t ON t.Id_Cliente = c.Id_Cliente
    LEFT JOIN Detalle_Tickets dt ON dt.Id_Det_Ticket = t.Id_Ticket
    LEFT JOIN Trenes_Recorridos_Estaciones tr ON tr.Id_Tren_Recorrido_Est = dt.Id_Tren_Recorrido_Est
    LEFT JOIN Recorridos r ON r.Nro_Linea = tr.Nro_Linea
    LEFT JOIN Estaciones es ON es.Id_Estacion = tr.Id_Estacion
    LEFT JOIN TrenesxVagones tv ON tv.Id_Trenes_Vagones = dt.Id_Trenes_Vagones
    LEFT JOIN Trenes tre ON tre.Id_Tren = tv.Id_Tren
    WHERE c.NroDoc = @NroDoc
);

GO

CREATE VIEW vw_Compras_Importantes
AS
SELECT
    f.Nro_Factura,
    f.Fecha,
    p.Nombre AS Proveedor,
    e.Apellido + ', ' + e.Nombre AS Empleado,
    COUNT(d.Id_Detalle_Factura) AS Cant_Facturas,
    SUM(d.Cantidad) AS Unidades,
    SUM(d.Cantidad * d.Precio) AS Total
FROM Facturas f
JOIN Detalles_Facturas d ON d.Nro_Factura = f.Nro_Factura
LEFT JOIN Proveedores p ON p.Id_Proveedor = f.Id_Proveedor
LEFT JOIN Empleados e ON e.Id_Empleado = f.Id_Empleado
WHERE
    p.Nombre  IN (SELECT p2.Nombre FROM Facturas f2
            JOIN Proveedores p2 ON p2.Id_Proveedor = f2.Id_Proveedor
                               GROUP BY p2.Nombre
                               HAVING COUNT(*) > 1)
GROUP BY f.Nro_Factura, f.Fecha, p.Nombre, e.Apellido, e.Nombre
HAVING SUM(d.Cantidad * d.Precio) > (SELECT AVG(Monto) FROM Facturas)

GO
